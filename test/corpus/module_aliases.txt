module sharing

extend type group
  relations
    define editor: [role#member] or owner
    define viewer: [role#member] or editor

    # Membership
    define member_admin: [role#member] or editor
    define member_viewer: [role#member] or member_admin or viewer

    define member_add: member_admin
    define member_list: member_viewer

extend type folder
  relations
    define editor: [role#member] or owner
    define viewer: [role#member] or editor or commenter

    # Sharing
    define share: owner or editor

extend type user
  relations
    define group_owner: [group]
    define folder_owner: [folder]
    define owner: group_owner or folder_owner
